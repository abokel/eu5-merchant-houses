# Merchant Houses on_actions (EU5 merges these with the vanilla ones of the same name)

# Every month, for every country (root = country)
monthly_country_pulse = {
	on_actions = { mh_monthly_country_pulse }
}

mh_monthly_country_pulse = {
	trigger = {
		OR = {
			government_type = government_type:corporation
			has_variable = mh_stake_holder
		}
	}
	effect = {
		if = {
			limit = { mh_is_corporation = yes }
			mh_monthly_corporation_effect = yes
		}
		if = {
			limit = { has_variable = mh_stake_holder }
			mh_monthly_host_effect = yes
		}
	}
}

# Winning an Enforce Charter war restores the charter (root = country, scope:winner, scope:loser, scope:war)
on_winning_war = {
	on_actions = { mh_on_winning_war }
}

mh_on_winning_war = {
	trigger = {
		scope:war = {
			casus_belli ?= casus_belli:mh_cb_enforce_charter
			is_an_attacker = scope:winner
		}
		scope:winner = { mh_is_corporation = yes }
	}
	effect = {
		scope:loser = { save_scope_as = mh_charter_loser }
		scope:winner = {
			mh_restore_charter_effect = yes
		}
	}
}
