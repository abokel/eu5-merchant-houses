# Merchant Houses - scripted triggers

# Country scope. Is this country a Corporation?
mh_is_corporation = {
	government_type = government_type:corporation
}

# Country scope. Can this country incorporate? Banks, the Hansa and merchant republics.
mh_can_incorporate = {
	NOT = { government_type = government_type:corporation }
	OR = {
		has_reform = government_reform:banking_country
		has_reform = government_reform:merchant_republic
		has_reform = government_reform:hanseatic_merchant_federation
	}
}

# Location scope. Where Presidencies (landed chartered company holdings) can be founded.
mh_presidency_zone_location = {
	OR = {
		sub_continent = sub_continent:south_asia
		sub_continent = sub_continent:south_east_asia
	}
}

# Country scope (a possible host). A real, landed country that a corporation can hold a charter in.
mh_valid_host = {
	country_type = location
	NOT = { government_type = government_type:corporation }
	exists = capital
}

# Interaction scope: scope:recipient (host) has granted scope:actor (corporation) a corporate charter.
mh_actor_holds_charter = {
	scope:recipient = {
		giving_scripted_relation = {
			type = relation_type:mh_corporate_charter
			target = scope:actor
		}
	}
}

# Interaction scope: scope:actor is the corporation holding the stake in scope:recipient.
mh_actor_is_stakeholder = {
	scope:recipient = {
		has_variable = mh_stake_holder
		var:mh_stake_holder = scope:actor
	}
}

# Interaction scope: scope:recipient's control by scope:actor is at least $VALUE$.
# Usage: mh_actor_control_at_least = { VALUE = 50 }
mh_actor_control_at_least = {
	mh_actor_is_stakeholder = yes
	scope:recipient = {
		has_variable = mh_control
		var:mh_control >= $VALUE$
	}
}

# Interaction scope: scope:actor's board confidence is at least $VALUE$.
mh_actor_board_at_least = {
	scope:actor = {
		has_variable = mh_board_confidence
		var:mh_board_confidence >= $VALUE$
	}
}

# Interaction scope: the rulers of scope:actor and scope:recipient share a dynasty.
mh_family_ties = {
	scope:actor.ruler ?= {
		has_dynasty = yes
		scope:recipient.ruler ?= {
			has_dynasty = yes
			dynasty = prev.dynasty
		}
	}
}

# Country scope (a host with a stake). Do the rulers of this host and its stakeholder share a dynasty?
mh_host_family_ties = {
	has_variable = mh_stake_holder
	ruler ?= {
		has_dynasty = yes
		prev.var:mh_stake_holder.ruler ?= {
			has_dynasty = yes
			dynasty = prev.dynasty
		}
	}
}

# Country scope (a host with a stake). Does it owe its stakeholder money?
mh_host_owes_stakeholder = {
	has_variable = mh_stake_holder
	any_loan = {
		owner ?= prev.var:mh_stake_holder
	}
}
