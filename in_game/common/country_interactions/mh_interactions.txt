# Merchant Houses - corporation interactions.
# scope:actor is always the corporation, scope:recipient the host country.

#######################################################
# Extend Credit: lend money to another country
#######################################################
mh_extend_credit = {
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS
	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = { mh_is_corporation = yes }
	}

	allow = {
		scope:actor = { gold >= 50 }
	}

	diplo_chance = {
		base = 15
		opinion = 0.2
		trust_in_actor = 0.2
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_loan
		}
		visible = {
			mh_valid_host = yes
			within_diplomatic_range = scope:actor
		}
	}

	effect = {
		scope:actor = {
			give_loan = {
				target = scope:recipient
				amount = {
					value = scope:recipient.monthly_income_trade_and_tax
					multiply = 6
					max = {
						value = scope:actor.gold
						multiply = 0.5
					}
				}
				interest = scope:recipient.modifier:bank_interest
				months = 60
			}
		}
		if = {
			limit = { mh_actor_is_stakeholder = yes }
			scope:recipient = { mh_change_control = { VALUE = 3 } }
		}
	}

	ai_will_do = {
		add = -10
		if = {
			limit = {
				scope:actor = { gold > 500 }
				mh_actor_is_stakeholder = yes
			}
			add = 30
		}
	}
}

mh_petition_trading_rights = {
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS
	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = { mh_is_corporation = yes }
	}

	allow = {
		custom_tooltip = {
			text = mh_board_needs_25_tt
			mh_actor_board_at_least = { VALUE = 25 }
		}
	}

	diplo_chance = {
		base = 0
		opinion = 0.2
		trust_in_actor = 0.2
	}

	accept = {
		if = {
			limit = { scope:recipient = { any_loan = { owner ?= scope:actor } } }
			add = {
				desc = "MH_DIPLOREASON_IN_DEBT_TO_THEM"
				value = 20
			}
		}
		if = {
			limit = { mh_family_ties = yes }
			add = {
				desc = "MH_DIPLOREASON_FAMILY_TIES"
				value = 25
			}
		}
		if = {
			limit = { mh_actor_control_at_least = { VALUE = 25 } }
			add = {
				desc = "MH_DIPLOREASON_CORPORATE_CONTROL"
				value = 15
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_trade
		}
		visible = {
			has_variable = mh_stake_holder
			var:mh_stake_holder = scope:actor
			giving_scripted_relation = {
				type = relation_type:mh_corporate_charter
				target = scope:actor
			}
		}
		enabled = {
			NOT = { has_variable = mh_right_build }
		}
	}

	effect = {
		scope:recipient = {
			set_variable = mh_right_build
			mh_change_control = { VALUE = 5 }
		}
		custom_tooltip = mh_petition_trading_rights_tt
	}

	ai_will_do = {
		add = 20
	}
}

#######################################################
# Absorb Host (control 100, or 75 with family ties)
#######################################################
mh_absorb_host = {
	type = diplomacy
	category = CATEGORY_INFLUENCE_ACTIONS
	ai_tick = monthly
	ai_tick_frequency = 12

	potential = {
		scope:actor = { mh_is_corporation = yes }
	}

	allow = {
		scope:actor = { at_war = no }
		custom_tooltip = {
			text = mh_board_needs_60_tt
			mh_actor_board_at_least = { VALUE = 60 }
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_influence
		}
		visible = {
			has_variable = mh_stake_holder
			var:mh_stake_holder = scope:actor
		}
		enabled = {
			at_war = no
			custom_tooltip = {
				text = mh_absorb_requirements_tt
				has_variable = mh_control
				OR = {
					AND = {
						var:mh_control >= 100
						is_subject_of = scope:actor
					}
					AND = {
						var:mh_control >= 75
						ruler ?= {
							has_dynasty = yes
							scope:actor.ruler ?= {
								has_dynasty = yes
								dynasty = prev.dynasty
							}
						}
					}
				}
			}
		}
	}

	effect = {
		custom_tooltip = mh_absorb_host_tt
		hidden_effect = {
			scope:recipient = { mh_prepare_absorption_effect = yes }
			scope:actor = {
				mh_change_board_confidence = { VALUE = -20 }
				mh_make_landed_effect = yes
				annex_country = { country = scope:recipient }
			}
		}
	}

	ai_will_do = {
		add = 50
	}
}

#######################################################
# Foreclose on Land: a debtor pays off every loan it owes us with a location
#######################################################
mh_foreclose_land = {
	type = diplomacy
	category = CATEGORY_ECONOMY_ACTIONS
	ai_tick = monthly
	ai_tick_frequency = 24

	potential = {
		scope:actor = { mh_is_corporation = yes }
	}

	allow = {
		scope:actor = { at_war = no }
		custom_tooltip = {
			text = mh_board_needs_30_tt
			mh_actor_board_at_least = { VALUE = 30 }
		}
	}

	diplo_chance = {
		base = -20
		opinion = 0.2
		trust_in_actor = 0.2
	}

	accept = {
		if = {
			limit = {
				scope:recipient = {
					any_loan = {
						owner ?= scope:actor
						count >= 2
					}
				}
			}
			add = {
				desc = "MH_DIPLOREASON_MANY_LOANS"
				value = 20
			}
		}
		if = {
			limit = {
				scope:recipient = {
					any_loan = {
						owner ?= scope:actor
						count >= 4
					}
				}
			}
			add = {
				desc = "MH_DIPLOREASON_MANY_LOANS"
				value = 20
			}
		}
		if = {
			limit = { scope:recipient = { gold < 0 } }
			add = {
				desc = "MH_DIPLOREASON_NEEDS_MONEY"
				value = 25
			}
		}
		if = {
			limit = { mh_actor_control_at_least = { VALUE = 50 } }
			add = {
				desc = "MH_DIPLOREASON_CORPORATE_CONTROL"
				value = 20
			}
		}
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		default_sort = will
		column = {
			data = country_diplomacy_info_loan
		}
		visible = {
			any_loan = { owner ?= scope:actor }
		}
		enabled = {
			at_war = no
		}
	}

	select_trigger = {
		looking_for_a = location
		interaction_source_list = {
			scope:recipient = {
				every_owned_location = {
					limit = { is_capital = no }
					add_to_list = source
				}
			}
		}
		target_flag = target
		name = "mh_select_foreclosure"
		none_available_msg_key = "mh_no_locations"
		column = {
			data = name
		}
		visible = {
			always = yes
		}
	}

	effect = {
		if = {
			limit = { exists = scope:target }
			custom_tooltip = mh_foreclose_land_tt
			hidden_effect = {
				scope:recipient = {
					every_loan = {
						limit = { owner ?= scope:actor }
						cancel_loan = this
					}
				}
				scope:actor = {
					mh_make_landed_effect = yes
					mh_change_board_confidence = { VALUE = 5 }
				}
				scope:target = {
					mh_take_location_effect = { CORP = scope:actor }
				}
			}
		}
		else = {
			custom_tooltip = mh_select_foreclosure
		}
	}

	ai_will_do = {
		add = 5
	}
}
