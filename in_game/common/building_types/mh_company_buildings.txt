# Merchant Houses - company buildings.
#
# Part 1: foreign buildings a corporation owns inside a host country.
#   mh_company_estate   - the corporation's share of the land (from a Land Grant petition)
#   mh_company_factory  - a factor's house (needs Trading Rights)
#   mh_company_barracks - company troops and levies (needs Levy Rights)
#
# Part 2: home-territory Presidency buildings for chartered companies in South and
# Southeast Asia. Same local effects as the vanilla trade company buildings.

#######################################################
# Part 1: foreign buildings in a host
#######################################################

# Granted land. Placed automatically by the Land Grant petition on the chosen location.
# When the company lands, these locations become its own.
mh_company_estate = {
	audio_tier = 3
	is_foreign = yes
	want_foreign_pop_created = no
	max_levels = 1
	pop_type = burghers
	category = trade_category
	employment_size = large_foreign_trade_employment
	pop_size_created = large_foreign_trade_employment
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	build_time = trade_company_build_time

	country_potential = {
		mh_is_corporation = yes
	}
	location_potential = {
		has_variable = mh_company_land
		var:mh_company_land = scope:actor
	}
	possible_production_methods = {
		satellite_trade_building_requirements
	}
	modifier = {
		scaled_gold_to_building_owner = 0.3
		merchant_power_from_building = 0.2
		local_crown_estate_power = -0.1
	}
	custom_tags = { trade_company_building }
}

# Factor's house: needs a charter and Trading Rights from the host.
mh_company_factory = {
	audio_tier = 3
	is_foreign = yes
	stronger_power_projection = no
	max_levels = 2
	pop_type = burghers
	own_or_overlord_relation_needed = mh_corporate_charter
	category = trade_category
	employment_size = foreign_trade_employment
	pop_size_created = foreign_trade_employment
	rural_settlement = no
	town = yes
	city = yes
	megalopolis = yes
	build_time = trade_post_build_time

	country_potential = {
		mh_is_corporation = yes
	}
	location_potential = {
		owner ?= {
			has_variable = mh_right_build
			has_variable = mh_stake_holder
			var:mh_stake_holder = scope:actor
		}
	}
	allow = {
		market = { in_trade_range_of = scope:actor }
	}
	possible_production_methods = {
		satellite_trade_building_requirements
	}
	modifier = {
		merchant_capacity_from_building = 0.4
		merchant_power_from_building = 0.25
		scaled_gold_to_building_owner = 0.15
	}
	custom_tags = { trade_company_building }
}

# Part 2: Presidency buildings (chartered companies, own land in South / Southeast Asia)
#######################################################

# Placed automatically in the first Presidency location
mh_presidency_headquarters = {
	audio_tier = 4
	is_foreign = no
	max_levels = estate_building_single_level
	pop_type = burghers
	category = trade_category
	employment_size = huge_foreign_trade_employment
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	expensive = yes
	build_time = trade_company_headquarters_time

	country_potential = {
		has_reform = government_reform:mh_chartered_company
	}
	location_potential = {
		mh_presidency_zone_location = yes
		has_variable = mh_presidency_seat
	}
	unique_production_methods = {
		mh_presidency_headquarters_requirements = {
			icon_type = building_type
			icon = trade_company_headquarters
			paper = 0.1
			category = building_maintenance
		}
	}
	modifier = {
		merchant_capacity_from_building = 0.3
		local_crown_estate_power = -0.2
	}
	raw_modifier = {
		local_monthly_development = 0.01
		local_monthly_prosperity = 0.001
		local_build_buildings_efficiency = 0.2
	}
	custom_tags = { trade_company_building }
}
